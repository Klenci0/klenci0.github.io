# 1. Inicie o git
git init

# 2. Adicione todos os arquivos
git add .

# 3. Crie o commit
git commit -m "feat: calculadora completa com operacoes basicas, porcentagem, raiz e potencia"

# 4. Defina a branch main
git branch -M main

# 5. Conecte ao seu repositório no GitHub
git remote add origin https://github.com/SEU-USUARIO/calculadora.git

# 6. Envie o código
git push -u origin main
